CREATE TABLE tags (
    id INT PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE
);

CREATE TABLE games (
    id INT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    tags INT[] -- Native array of Tag IDs (or use JSONB)
);

CREATE TABLE weeksheet (
    id SERIAL PRIMARY KEY,
    week_date DATE NOT NULL,          -- Date of the Tuesday for that week
    rank INT NOT NULL,                -- 1 to 100
    game_id INT NOT NULL REFERENCES games(id) ON DELETE CASCADE,
    price DECIMAL(10, 2) NOT NULL,
    rating DECIMAL(10, 2),
    
    -- Prevents duplicate rank entries in the same week
    CONSTRAINT unique_week_rank UNIQUE (week_date, rank),
    -- Prevents the same game from being listed twice in the same week
    CONSTRAINT unique_week_game UNIQUE (week_date, game_id)
);

-- Fast lookup indexes
CREATE INDEX idx_weeksheet_date ON weeksheet (week_date);
CREATE INDEX idx_weeksheet_game ON weeksheet (game_id);
